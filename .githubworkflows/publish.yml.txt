# Название workflow
name: Deploy to GitHub Pages

# Запускать при пуше в ветку main (или master)
on:
  push:
    branches: ["main"]
  workflow_dispatch:

# Устанавливаем права для деплоя
permissions:
  contents: read
  pages: write
  id-token: write

# Задача
jobs:
  deploy:
    environment:
      name: github-pages
      url: ${{ steps.deployment.outputs.page_url }}
    runs-on: ubuntu-latest
    steps:
      - name: Checkout code
        uses: actions/checkout@v4

      - name: Setup Pages
        uses: actions/configure-pages@v4

      - name: Upload artifact
        uses: actions/upload-pages-artifact@v3
        with:
          # Укажите здесь вашу папку с сайтом, например: mainapp, authapp или .
          path: 'mainapp/templates/mainapp'

      - name: Deploy to GitHub Pages
        id: deployment
        uses: actions/deploy-pages@v4
